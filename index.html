    function doGet(e) {
  var rootName = (e && e.parameter && e.parameter.rootFolder) || 'TEEM BUSINESS PORTAL';
  var rootIt = DriveApp.getFoldersByName(rootName);
  if (!rootIt.hasNext()) {
    return ContentService.createTextOutput(JSON.stringify({ ok: true, empty: true })).setMimeType(ContentService.MimeType.JSON);
  }
  var root = rootIt.next();
  var bIt = root.getFoldersByName('Backups');
  if (!bIt.hasNext()) {
    return ContentService.createTextOutput(JSON.stringify({ ok: true, empty: true })).setMimeType(ContentService.MimeType.JSON);
  }
  var bFolder = bIt.next();
  var fIt = bFolder.getFilesByName('TEEM_AutoBackup_Latest.json');
  if (!fIt.hasNext()) {
    return ContentService.createTextOutput(JSON.stringify({ ok: true, empty: true })).setMimeType(ContentService.MimeType.JSON);
  }
  var raw = fIt.next().getBlob().getDataAsString();
  return ContentService.createTextOutput(raw).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents || '{}');
  var rootName = data.rootFolder || 'TEEM BUSINESS PORTAL';
  var rootIt = DriveApp.getFoldersByName(rootName);
  var rootAlreadyExisted = rootIt.hasNext();
  var root = rootAlreadyExisted ? rootIt.next() : DriveApp.createFolder(rootName);
  var sectors = ['Commercial','Industrial','Medical','Institutional','Agricultural Waste','Construction Waste','Municipal','Monthly Reports','Backups'];
  for (var i = 0; i < sectors.length; i++) {
    if (!root.getFoldersByName(sectors[i]).hasNext()) root.createFolder(sectors[i]);
  }
  if (data.action === 'pull_state') {
    return doGet({ parameter: { rootFolder: rootName } });
  }
  if ((data.action === 'save_request' || data.action === 'save_monthly_report') && data.pdfBase64) {
    var catName = data.categoryFolder || (data.action === 'save_monthly_report' ? 'Monthly Reports' : 'Commercial');
    var catIt = root.getFoldersByName(catName);
    var catFolder = catIt.hasNext() ? catIt.next() : root.createFolder(catName);
    var fileName = data.fileName || ('TEEM_' + catName + '_Request.pdf');
    var existingFileIt = catFolder.getFilesByName(fileName);
    if (existingFileIt.hasNext()) {
      var existingFile = existingFileIt.next();
      return ContentService.createTextOutput(JSON.stringify({ ok: true, alreadyExists: true, message: 'Success: "' + fileName + '" already exists in ' + rootName + '/' + catName + '.', folder: rootName + '/' + catName, name: existingFile.getName(), url: existingFile.getUrl() })).setMimeType(ContentService.MimeType.JSON);
    }
    var bytes = Utilities.base64Decode(data.pdfBase64);
    var blob = Utilities.newBlob(bytes, 'application/pdf', fileName);
    var file = catFolder.createFile(blob);
    return ContentService.createTextOutput(JSON.stringify({ ok: true, alreadyExists: false, folder: rootName + '/' + catName, name: file.getName(), url: file.getUrl() })).setMimeType(ContentService.MimeType.JSON);
  }
  var snap = data.snapshot || data;
  if (snap && snap.state && snap.state.monthlyReports) {
    var mrFolder = root.getFoldersByName('Monthly Reports').next();
    var months = Object.keys(snap.state.monthlyReports);
    for (var m = 0; m < months.length; m++) {
      var period = months[m];
      var mrFile = 'TEEM_Monthly_Report_' + period.replace(/\s+/g, '_') + '.json';
      var mrBody = JSON.stringify(snap.state.monthlyReports[period], null, 2);
      var mrEx = mrFolder.getFilesByName(mrFile);
      if (mrEx.hasNext()) mrEx.next().setContent(mrBody);
      else mrFolder.createFile(mrFile, mrBody, MimeType.PLAIN_TEXT);
    }
  }
  var bFolder = root.getFoldersByName('Backups').next();
  var bName = 'TEEM_AutoBackup_Latest.json';
  var existing = bFolder.getFilesByName(bName);
  var content = JSON.stringify(snap, null, 2);
  if (existing.hasNext()) { existing.next().setContent(content); }
  else { bFolder.createFile(bName, content, MimeType.PLAIN_TEXT); }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, folderAlreadyExisted: rootAlreadyExisted, message: rootAlreadyExisted ? ('Success: Folder "' + rootName + '" already exists!') : ('Success: Folder "' + rootName + '" created!'), folder: rootName + '/Backups', savedAt: new Date().toISOString() })).setMimeType(ContentService.MimeType.JSON);
}
